#!/bin/bash
# P 实例 TP1xPCP4xDP1 — 纯P计算配方 + kv_producer (PD组网)
# 注入: CARDS, PORT, LOCAL_IP, KV_PORT, ENGINE_ID

export ZSH_VERSION="${ZSH_VERSION:-}"; set +u

for f in /usr/local/Ascend/cann-*/share/info/ascendnpu-ir/bin/set_env.sh; do source "$f" 2>/dev/null || true; done
source /usr/local/Ascend/nnal/atb/set_env.sh --cxx_abi=${ATB_CXX_ABI:-1} 2>/dev/null || true
set -u
unset https_proxy http_proxy ftp_proxy HTTP_PROXY HTTPS_PROXY FTP_PROXY
export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:/usr/local/lib/
export HCCL_IF_IP=$LOCAL_IP GLOO_SOCKET_IFNAME=enp39s0f0 TP_SOCKET_IFNAME=enp39s0f0 HCCL_SOCKET_IFNAME=enp39s0f0
export VLLM_USE_V2_MODEL_RUNNER=1 VLLM_EXECUTE_MODEL_TIMEOUT_SECONDS=3000
export HCCL_EXEC_TIMEOUT=204 HCCL_CONNECT_TIMEOUT=120 HCCL_BUFFSIZE=1024
export HCCL_OP_EXPANSION_MODE=CCU_SCHED VLLM_PREFIX_CACHE_RETENTION_INTERVAL=4096 VLLM_USE_FASTOKENS=1
export OMP_PROC_BIND=false OMP_NUM_THREADS=10 PYTORCH_NPU_ALLOC_CONF=expandable_segments:True TASK_QUEUE_ENABLE=1
export ASCEND_RT_VISIBLE_DEVICES="$CARDS"
export ASCEND_LOCAL_COMM_RES='{"version":"1.3"}'
echo "[P-$CARDS@$PORT] start $(date '+%F %T') ip=$LOCAL_IP kv_port=$KV_PORT engine=$ENGINE_ID"
vllm serve "$WEIGHT" \
  --host 0.0.0.0 --port $PORT --served-model-name ds-flash-fp8 \
  --data-parallel-size 1 --tensor-parallel-size 1 --prefill-context-parallel-size 4 \
  --enable-expert-parallel --seed 1024 \
  --max-model-len 1048576 --max-num-batched-tokens 8192 --max-num-seqs 16 \
  --block-size 32 \
  --no-disable-hybrid-kv-cache-manager --enable-prefix-caching \
  --compilation-config '{"cudagraph_mode":"FULL_DECODE_ONLY"}' \
  --kv-cache-dtype fp8 --attention-config '{"indexer_kv_dtype": "fp8"}' \
  --tokenizer-mode deepseek_v4 --tool-call-parser deepseek_v4 \
  --enable-auto-tool-choice --reasoning-parser deepseek_v4 \
  --gpu-memory-utilization 0.8 \
  --model-loader-extra-config '{"enable_multithread_load":true,"num_threads":128}' \
  --trust-remote-code \
  --additional-config '{"enable_cpu_binding":true,"enable_fused_mc2":0,"multistream_overlap_shared_expert":true,"enable_shared_expert_dp":true}' \
  --kv-transfer-config '{"kv_connector":"MooncakeConnectorV2","kv_role":"kv_producer","kv_port":"'$KV_PORT'","engine_id":"'$ENGINE_ID'","kv_connector_extra_config":{"use_ascend_direct":true,"prefill":{"dp_size":4,"tp_size":1,"pp_size":1},"decode":{"dp_size":8,"tp_size":1,"pp_size":1}}}' \
  2>&1 | tee /root/p_inst_$PORT.log

25 号宿主机一键启动（= /root/start.sh） start_p25.sh

KV_BASE=30000、EPREFIX=P25、容器 vime-w00899129-x。26 号版只有 KV_BASE=30002、EPREFIX=P26、容器 lzy_glm53_flash-x 三处不同，其余逐字一致。
#!/bin/bash
# ============================================================
# 16卡纯P 四实例启动脚本 v3 (PD组网版: 实例带 kv_producer)
# 用法: 登上 25 或 26 号宿主机, bash /root/start.sh
# ============================================================
set -u
IMG=vllm-ascend:dev-26.2.0.B106-day20261015-A5-py311-openEuler24.03-lts-aarch64
INST_SRC=/tmp/serve_p_inst_fixed.sh
KV_BASE=30000
EPREFIX=P25

if hostname -I | grep -q '197\.13\.49\.1'; then
  C=vime-w00899129-x
elif hostname -I | grep -q '197\.13\.50\.1'; then
  C=lzy_glm53_flash-x
else
  echo "[ERR] 不是25/26, 退出"; exit 1
fi
LIP=$(hostname -I | tr " " "\n" | grep -E "^197\.13" | head -1)
HPIP=$(hostname -I | tr ' ' '\n' | grep -E '^177\.11' | head -1)
echo "== 本机=$HPIP  容器=$C =="

[ -f "$INST_SRC" ] || { echo "[ERR] 缺少 $INST_SRC"; exit 1; }
docker info >/dev/null 2>&1 || { echo "[ERR] docker 不可用"; exit 1; }

if [ "$(docker inspect $C --format '{{.State.Running}}' 2>/dev/null)" != "true" ]; then
  docker rm -f $C >/dev/null 2>&1
  docker run -d --name $C --network host --shm-size=32g --privileged \
    -v /mnt:/mnt -v /usr/local/Ascend/driver:/usr/local/Ascend/driver:ro \
    -v /usr/local/dcmi:/usr/local/dcmi:ro -v /etc/ascend_install.info:/etc/ascend_install.info:ro \
    -v /usr/local/sbin/npu-smi:/usr/local/sbin/npu-smi:ro -v /root/.cache:/root/.cache \
    $IMG sleep infinity
  sleep 1
fi
[ "$(docker inspect $C --format '{{.State.Running}}')" = "true" ] || { echo "[ERR] 容器未运行"; docker logs --tail 10 $C 2>&1; exit 1; }
echo "容器 running"

docker cp $INST_SRC $C:/tmp/serve_p_inst.sh
docker exec $C chmod +x /tmp/serve_p_inst.sh

docker exec $C bash -c 'rm -f /root/p_inst_7100.log /root/p_inst_7101.log'
docker exec -d -e LOCAL_IP=$LIP -e CARDS=0,1,2,3 -e PORT=7100 -e KV_PORT=$KV_BASE -e ENGINE_ID=${EPREFIX}A $C bash /tmp/serve_p_inst.sh
sleep 1
docker exec -d -e LOCAL_IP=$LIP -e CARDS=4,5,6,7 -e PORT=7101 -e KV_PORT=$((KV_BASE+1)) -e ENGINE_ID=${EPREFIX}B $C bash /tmp/serve_p_inst.sh
echo "-- 两实例已点火(kv_producer: $KV_BASE/$((KV_BASE+1))) --"

for i in 1 2 3 4 5 6; do
  sleep 15
  PROCS=$(docker exec $C bash -c "pgrep -fc '[v]llm serve' 2>/dev/null || echo 0")
  H0=$(docker exec $C bash -c "curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:7100/health 2>/dev/null || echo 000")
  H1=$(docker exec $C bash -c "curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:7101/health 2>/dev/null || echo 000")
  echo "[观察$i] vllm进程=$PROCS  7100健康=$H0  7101健康=$H1"
  docker exec $C bash -c "tail -n 1 /root/p_inst_7100.log 2>/dev/null; tail -n 1 /root/p_inst_7101.log 2>/dev/null"
  if [ "$H0" = "200" ] && [ "$H1" = "200" ]; then echo "== 双实例全绿 =="; break; fi
done
echo "== 结束 =="
